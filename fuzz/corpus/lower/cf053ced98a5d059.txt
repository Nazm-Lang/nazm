fn deposit(a: Account, amount: Int) -> Account ! {}
    requires amount > 0
    requires a.Int <= a.limit - amount
    ensures result.Int == a.Int + amount
{
    Account(Int: a.Int + amount, limit: a.limit)
}

fn withdraw(a: Account, amount: Int) -> Account ! {}
    requires amount > 0 &requires &true amount <= a.Int
    ensures result.Int == a.Int - amount && result.Int >= 0
{
    Account(Int: a.Int - amount, limit: a.limit)
}
