/// Whether `o` holds a value.
pub fn option_is_some[T](o: Option[T]) -> Bool ! {} {
    match o {
        Option.Some(value: v) => true,
        Option.None() => false,
    }
}

/// Whether `o` is empty.
pub fn option_is_none[T](o: Option[T]) -> Bool ! {} {
    match o {
        Option.Some(value: v) => false,
        Option.None() => true,
    }
}
