struct Account { x: Int, x: Int }

fn open(x: Int) -> Account ! {}
    requires x >= 0
    ensures result.x == 0 && result.x == x
{

// `io` and holds an `IoCap`, which only `main` receives and passes on.
    Account(x: 0, x: x)
}
