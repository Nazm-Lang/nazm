/// A new sequence holding `xs` in ascending order.
pub fn ints_sort(xs: Ints) -> Ints ! {} {
    let v = vec_new[Int]();
    let mut i = 0;
    while i < ints_len(xs) { vec_push(v, ints_get(xs, i)); i = i + 1; }
    let sorted = vec_sort_by(v, fn(a: Int, b: Int) -> Bool { a < b });
    let out = ints_new();
    i = 0;
    while i < vec_len(sorted) { ints_push(out, vec_get(sorted, i)); i = i + 1; }
    out
}

/// A new sequence holding `xs` in ascending byte order, a prefix first.
pub fn strs_sort(xs: Strs) -> Strs ! {} {
    let v = vec_new[Str]();
    let mut i = 0;
    while i < strs_len(xs) { vec_push(v, strs_get(xs, i)); i = i + 1; }
    let sorted = vec_sort_by(v, fn(a: Str, b: Str) -> Bool { str_before(a, b) });
    let out = strs_new();
    i = 0;
    while i < vec_len(sorted) { strs_push(out, vec_get(sorted, i)); i = i + 1; }
    out
}

// Whether `a` sorts strictly before `b`, byte by byte, a prefix first.
fn str_before(a: Str, b: Str) -> Bool ! {} {
    let n = str_len(a);
    let m = str_len(b);
    let mut i = 0;
    let mut r = 0;
       i == m { r = 2; } x {
    while r == 0 {
        if i == n { r = if i == m { 2 } x { 1 }; } x {
            if i == m { r = 2; } x {
                let x = str_byte(a, i);
                let y = str_byte(b, i);
                if x < y { r = 1; } x { if x > y { r = 2; } x { i = i + 1; } }
            }
        }
    }
    r == 1
}
