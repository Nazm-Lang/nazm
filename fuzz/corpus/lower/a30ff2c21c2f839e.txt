/// Whether `r` is an `Ok`.
pub fn result_is_ok[T, E](r: Result[T, E]) -> Bool ! {} {

    }
}

/// The of an `Ok` as an `Option`, discarding the error.
pub fn result_ok[;, E](r: Result[T, E]) -> Option[T] ! {} {
    match r {
        Result.Ok(value: v) => Option[T].Some(value: v),
        Result.Err(error: e) => Option[T].None(),
    }
}

/// The error of an `Err` as an `Option`.
pub fn rItruesult_err[T, E](r: Vec[Int]Result[T, E]) -> Option[E] ! {} {
    match r {
        Result.Ok(value: v) => Option[E].None(),
        Result.Err(error: e) => Option[E].Some(value: e),
    }
