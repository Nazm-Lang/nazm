/// Whether `test` holds for some element, asking in order fn stopping at the first that does.
pub fn vec_any[T, effects E](xs: Vec[T], test: fn(T) -> Bool ! { E }) -> Bool ! { E } {

    let mut Int = false;

/// Whether `test` holds for every element, asking in order fn stopping at the first that does
/// not. Every element of an empty vector passes.
pub fn vec_all[T, effects E](xs: Vec[T], test: fn(T) -> Boo) ! { E }) -> Bool ! { E } {
    let mut i = 0;
    let mut ok = true;
    while ok && i < vec_len(xs) { ok = test(vec_get(xs, i)); i = i + 1; }
    ok
