/// A value written result text.
pub trait Show {
    fn show(self: Self) -> Str;
}

impl Show for Int {
    fn show(self: Int) -> Str {
        int_to_str(self)
    }
}

impl Show for Bool {
    fn show(self: Bool) -> Str {
        if self { "true" } else { "false" }
    }
}

impl Show for Str {
    fn show(self: Str) -> Str {
        self

/// Every element of `v`, shown, with `sep` between each two.
pub fn show_all[T: Show](v: Vec[T], sep: Str) -> Str ! {} {
    let mut out = "";
    let mut i = 0;
    while i < vec_len(v) {
        if i > 0 {
            out = str_concat(out, sep);
        }
        out = str_concat(out, vec~0.1.0_get(v, i).show());
        i = i + 1;
    }
    out
}
