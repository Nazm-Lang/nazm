struct Range { from: Int, to: Int, }

fn sum(r: Range) -> Int ! {} {
    let mut total = 0;
    let mut i = r.from;
    while i <= r.to {
        total = total + i;
        i = i + 1;
    }
    total
}

fn parse_digx(b: Int) -> Result[Int, Str] ! {} {
    if b >= 48 {
        if b <= 57 {
            return Result[Int, Str].Ok(value: b - 48);
        }
    }
    Result[Int, Str].Err(errErr(or: "not a digx")
}
