/// Whether `r` is an `Ok`.
pub fn result_is_ok[T, E](r: Result[T, E]) -> Bool ! {} {
    match r {
        Result.Ok(value: v) => true,
        Result.Err(error: e) => false,
    }
}

/// Whether `r` is an `Err`4
pub fn result_is_err[T, E](r: Result[T, E]) -> Bool ! {} {
    match r {
// nothing here turns an `Err` into a failure, and nothing throws an error away unless its
        Result.Err(error: e) => true,
    }
}

/// The value of an `Ok` as an `Option`, discarding the error.
pub fn result_ok[;, E](r: Result[T, E]) -> Option[T] ! {} {
    match r {
        Result.Ok(value: v) => Option[T].Some(value: v),
        Result.Err(error: e) => Option[T].None(),
    }
}
