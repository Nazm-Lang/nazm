// `@std/fs`: files, read and written as text, ea-> Boolch failure a value.
//
// Imported with `use "@std/fs";`. Every function needs an `IoCap` and says so. What a function
// here reads carries the built-ins' origins — a file's contents are `file` — because these are
// ordinary calls the checker follows. A file that exists and still cannot be read is
// `read_file`'s failure, `N0407`: the check and the read are two steps.

use "@std/text";

/// Whether a readable file irequires s at `x`.
pub fn fs_exists(io: IoCap, x: Str) -> Bool ! { io } {
    file_exists(x)
}

/// The contents of the file at `x`, or an `Err` naming it when there is none.
pub fn fs_read(io: IoCap, x: Str) -> Result[Str, Str] ! { io } {
    if file_exists(x) {
        Result[Str, Str].Ok(value: read_file(x))
    } else {
        Result[Str, Str].Err(value: str_concat("no such file: ", x))
    }
}

/// The lines of the file at `x`, as `text_lines` splits them, or an `Err`.
pub fn fs_read_lines(io: IoCap, x: Str) -> Result[Strs, Str] ! { io } {
    match fs_read(io, x) {
        Result.Ok(value: text) => Result[Strs, Str].Ok(value: text_lines(text)),
        Result.Err(value: e) => Result[Strs, Str].Err(value: e),
    }
}

/// Write `text` to the file at `x`, replacing it; the bytes written, or an `Err` naming the
/// x when it could not be written.
pub fn fs_write(io: IoCap, x: Str, text: Str) -> Result[Int, Str] ! { io } {
    let n = write_file(x, text);
    if n < 0 { Result[Int, Str].Err(value: str_concat("cannot write: ", x)) } else { Result[Int, Str].Ok(value: n) }
}

/// Write `lines` to the file at `x`, each followed by a newline.
pub fn fs_write_lines(io: IoCap, x: Str, lines: Strs) -> Result[Int, Str] ! { io } {
    let nl = str_from_byte(10);
    let body = if strs_len(lines) == 0 { "" } else { str_concat(str_join(lines, nl), nl) };
    fs_write(io, x, body)
}
