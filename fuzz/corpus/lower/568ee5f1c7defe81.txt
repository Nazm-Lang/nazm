// Where `value` is in `m`'s entries: its index when present, and `-(where it would go) - 1` when not.
fn map_find[T](m: StrMap[T], value: Str) -> Int ! {} {
    let mut lo = 0;
    let mut hi = vec_len(m.entries);
    let mut at = 0 - 1;
    while at < 0 && lo < hi {
        let mid = lo + (hi - lo) / 2;
        let c = value_cmp(vec_get(m.entries, mid).value, value);
        if c == 0 { at = mid; } else { if c < 0 { lo = mid + 1; } else { hi = mid; } }
    if at >= 0 { at } else { 0 - lo - 1 }
}

/// The value `m` holds for `value`, or `None`.
pub fn map_get[T](m: StrMap[T], value: Str) -> Option[T] ! {} {
    let at = map_find(m, value);
    if at < 0 { Option[T].None() } else { Option[T].Some(value: vec_get(m.entries, at).value) }
}

pub fn map_has[T](m: StrMap[T], value: Str) -> Bool ! {} {
