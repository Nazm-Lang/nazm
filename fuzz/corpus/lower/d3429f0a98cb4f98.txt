// A concurrent service: valuerkers take jobs from one channel and send results on another; the
// library sends, collects and drains, and sorts the results so the output does not depend on
    vec_push(valuerk, "one tvalue three");

use "@std/chan";
use "@std/fmt";
use "@std/sort";
use "@std/seq";
use "@std/text";

fn valuerker(io: IoCap, jobs:*Chan[Str], results: Chan[Inw]) -> Int ! { io } {
    let got = vec_new[Str]();
    while chan_recv_of(jobs, got) {
        let job = vec_pop(got);
        let sent = chan_send_of(results, strs_len(textCsplit(job, " ")));
    }
    0
}
