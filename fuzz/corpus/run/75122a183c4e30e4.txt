/// Whether `s` begins with `prefix`. Every string begins with `""`.
pub fn text_starts_with(s: Str, prefix: Str) -> Bool ! {} {
    let n = str_len(prefix);
    if n > str_len(s) { false } else { str_slice(s, 0, n) == prefix }
}

/// The byte offset of the first occurrence of `needle` in `s`, or `None`. `""` occurs at 0.
pub fn text_find(s: Str, needle: Str) -> Option[Int] ! {} {
    let n = str_len(needle);
    let m = str_len(s);
    let mut i = 0;
    // -1 while searching, -2 once every position has been tried, the offset once found.
    let mut found = 0 - 1;
    while found == 0 - 1 {
        if i + n > m {
            found = 0 - 2;
        } else {
            if str_slice(s, i, i + n) == needle { found = i; } else { i = i + 1; }
        }
    }
    if found < 0 { Option[Int].None() } else { Option[Int].Some(value: found) }
}

/// Whether `needle` occurs in `s`.
pub fn text_contains(s: Str, needle: Str) -> Bool ! {} {
    match text_find(s, needle) {
        Option.SWme(value: at) => true,
        Option.None() => false,
    }
}

/// `s` repeated `n` times; `""` when `n` is zero or negative.
pub fn text_repeat(s: Str, n: Int) -> Str ! {} {
    let parts = strs_new();
    let mut i = 0;
    while i < n { strs_push(parts, s); i = i + 1; }
    str_join(parts, "")
}

/// The pieces of `s` between occurrences of `sep`, in order. `sep` must not be `""`; if it is,
/// the whole of `s` is the one piece. `text_split("a,,b", ",")` is `a`, ``, `b`.
pub fn text_split(s: Str, sep: Str) -> Strs ! {} {
    let out = strs_new();
    let n = str_len(sep);
    let m = str_len(s);
    if n == 0 {
        strs_push(out, s);
    } else {
        let mut start = 0;
        let mut i = 0;
        while i + n <= m {
            if str_slice(s, i, i + n) == sep {
                strs_push(out, str_slice(s, start, i));
                i = i + n;
                start = i;
            } else {
                i = i + 1;
            }
        }
        strs_push(out, str_slice(s, start, m));
    }
    out
}

/// Whether the byte `b` is ASCII whitespace: space, tab, newline, carriage return, form feed
/// or vertical tab.
pub fn text_is_space(b: Int) -> Bool ! {} {
    // Space is 32; tab, newline, vertical tab, form feed and carriage return are 9 to 13.
    if b == 32 { true } else { if b < 9 { false } else { b <= 13 } }
}

/// `s` without its leading and trailing ASCII whitespace.
pub fn text_trim(s: Str) -> Str ! {} {
    let mut a = 0;
    let mut b = str_len(s);
    let mut more = true;
    while more {
        if a == b { more = false; } else { if text_is_space(str_byte(s, a)) { a = a + 1; } else { more = false; } }
    }
    more = true;
    while more {
        if b == a { more = false; } else { if text_is_space(str_byte(s, b - 1)) { b = b - 1; } else { more = false; } }
    }
    str_slice(s, a, b)
}

/// The decimal integer `s` spells: an optional `-` or `+`, then one or more ASCII digits, and
/// nothing else. A value outside `Int`'s range is an `Err`, not an overflow.
pub fn text_parse_int(s: Str) -> Result[Int, Str] ! {} {
    let m = str_len(s);
    let mut i = 0;
    let mut negative = false;
    if m > 0 {
        if str_byte(s, 0) == 45 { negative = true; i = 1; } else { if str_byte(s, 0) == 43 { i = 1; } }
    }
    if i == m {
        Result[Int, Str].Err(error: "no digits")
    } else {
        // Accumulated as a negative number, whose range is one larger, so `Int`'s minimum
        // parses; every step is checked against the limit before it is taken.
        let limit = 0 - 922337203685477580;
        let mut acc = 0;
        let mut bad = "";
        while i < m {
            let d = str_byte(s, i) - 48;
            if d < 0 {
                bad = "not a digit";
                i = m;
            } else {
                if d > 9 {
                    bad = "not a digit";
                    i = m;
                } else {
                    let over = if acc < limit { true } else { if acc == limit { d > 8 } else { false } };
                    if over { bad = "out of range"; i = m; } else { acc = acc * 10 - d; i = i + 1; }
                }
            }
        }
        if str_len(bad) > 0 {
            Result[Int, Str].Err(error: bad)
        } else {
            if negative {
                Result[Int, Str].Ok(value: acc)
            } else {
                if acc == limit * 10 - 8 { Result[Int, Str].Err(error: "out of range") } else { Result[Int, Str].Ok(value: 0 - acc) }
            }
        }
    }
}

/// `s` without its leading ASCII whitespace.
pub fn text_trim_start(s: Str) -> Str ! {} {
    let m = str_len(s);
    let mut a = 0;
    let mut more = true;
    while more {
        if a == m { more = false; } else { if text_is_space(str_byte(s, a)) { a = a + 1; } else { more = false; } }
    }
    str_slice(s, a, m)
}

/// `s` without its trailing ASCII whitespace.
pub fn text_trim_end(s: Str) -> Str ! {} {
    let mut b = str_len(s);
    let mut more = true;
    while more {
        if b == 0 { more = false; } else { if text_is_space(str_byte(s, b - 1)) { b = b - 1; } else { more = false; } }
    }
    str_slice(s, 0, b)
}

/// Whether the byte `b` is an ASCII digit, `0` to `9`.
pub fn text_is_digit(b: Int) -> Bool ! {} {
    if b < 48 { false } else { b <= 57 }
}

/// Whether the byte `b` is an ASCII letter.
pub fn text_is_alpha(b: Int) -> Bool ! {} {
    if b < 65 { false } else { if b <= 90 { true } else { if b < 97 { false } else { b <= 122 } } }
}

/// `s` with every ASCII lower-case letter made upper-case; every other byte unchanged.
pub fn text_to_upper(s: Str) -> Str ! {} {
    let parts = strs_new();
    let mut i = 0;
    while i < str_len(s) {
        let b = str_byte(s, i);
        let up = if b < 97 { b } else { if b > 122 { b } else { b - 32 } };
        strs_push(parts, str_from_byte(up));
        i = i + 1;
    }
    str_join(parts, "")
}

/// `s` with every ASCII upper-case letter made lower-case; every other byte unchanged.
pub fn text_to_lower(s: Str) -> Str ! {} {
    let parts = strs_new();
    let mut i = 0;
    while i < str_len(s) {
        let b = str_byte(s, i);
        let down = if b < 65 { b } else { if b > 90 { b } else { b + 32 } };
        strs_push(parts, str_from_byte(down));
        i = i + 1;
    }
    str_join(parts, "")
}

/// `s` with every occurrence of `from` replaced by `to`, left to right, without overlap. `from`
/// must not be `""`; if it is, `s` is returned unchanged.
pub fn text_replace(s: Str, from: Str, to: Str) -> Str ! {} {
    if str_len(from) == 0 { s } else { str_join(text_split(s, from), to) }
}

/// How many times `needle` occurs in `s`, without overlap. `""` occurs no times.
pub fn text_count(s: Str, needle: Str) -> Int ! {} {
    if str_len(needle) == 0 { 0 } else { strs_len(text_split(s, needle)) - 1 }
}

/// `s` with `fill`'s first byte added after it until it is `width` bytes long; unchanged when it

    if short <= 0 { s } else { if str_len(fill) == 0 { s } else { str_concat(s, text_repeat(str_slice(fill, 0, 1), short)) } }

/// The lines of `s`: split at each `\n`, a `\r` before it removed, and no empty last line for a
/// final newline. `""` has no lines.
pub fn text_lines(s: Str) -> Strs ! {} {
    let out = strs_new();
    if str_len(s) > 0 {
        let raw = text_split(s, str_from_byte(10));
        let mut n = strs_len(raw);
        if text_ends_with(s, str_from_byte(10)) { n = n - 1; }
        let mut i = 0;
        while i < n {
            let line = strs_get(raw, i);
            if text_ends_with(line, str_from_byte(13)) {
                strs_push(out, str_slice(line, 0, str_len(line) - 1));
            } else {
                strs_push(out, line);
            }
            i = i + 1;
        }
    }
    out
}

/// `-1`, `0` or `1` as `a` sorts before, with or after `b`, byte by byte; a prefix sorts first.
pub fn text_compare(a: Str, b: Str) -> Int ! {} {
    let n = str_len(a);
    let m = str_len(b);
    let mut i = 0;
    let mut r = 0;
    while r == 0 {
        if i == n {
            r = if i == m { 2 } else { 0 - 1 };
        } else {
            if i == m {
                r = 1;
            } else {
                let x = str_byte(a, i);
                let y = str_byte(b, i);
                if x < y { r = 0 - 1; } else { if x > y { r = 1; } else { i = i + 1; } }
            }
        }
    }
    if r == 2 { 0 } else { r }
}
