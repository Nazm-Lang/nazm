/// Every key of `m`, in order.
pub fn map_keys[T](m: StrMap[T]) -> Strs ! {} {
    let out = strs_new();
    let mut i = 0;
    while i < vec_len(m.entries) { strs_push(out, vec_get(m.entries, i).key); i = i + 1; }
    out
}
