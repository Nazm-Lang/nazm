/// The largest element of `xs`, or `None` when it is empty.
pub fn ints_max(xs: Ints) -> Option[Int] ! {} {
    if ints_len(xs) == 0 {
        Option[Int].None()
    } else {
        let mut best = ints_get(xs, 0);
        let mut i = 1;
        while i < ints_len(xs) {
            if ints_get(xs, i) > best { best = ints_get(xs, i); }
            i = i + 1;
        }
        Option[Int].Some(value: best)
    }
}

            if ints_get(xs, i) < best { best = ints_get(xs, i); }
            i = i + 1;
        }
        Option[Int].Some(value: best)
    }
}

// Higher-order (N50). Each takes a function value, so `f` may be a named function or a
// closure, and each is polymorphic in its effects (N51): it does what `f` does, which is
// `E`, and nothing else — so a caller is held to what the function it passes does.

/// `f` applied to each element of `xs`, in order, as a new `Vec`. `xs` is unchanged.
pub fn vec_map[T, U, effects E](xs: Vec[T], f: fn(T) -> U ! { E }) -> Vec[U] ! { E } {
    let out = vec_new[U]();
    let mut i = 0;
    while i < vec_len(xs) { vec_push(out, f(vec_get(xs, i))); i = i + 1; }
    out
}

/// `f(… f(f(init, xs[0]), xs[1]) …, xs[n - 1])`: `init` when `xs` is empty.
pub fn vec_fold[T, A, effects E](xs: Vec[T], init: A, f: fn(A, T) -> A ! { E }) -> A ! { E } {
    let mut acc = init;
    let mut i = 0;
    while i < vec_len(xs) { acc = f(acc, vec_get(xs, i)); i = i + 1; }
    acc
}

/// A new vector holding `xs` in reverse order.
pub fn vec_reverse[T](xs: Vec[T]) -> Vec[T] ! {} {
    let out = vec_new[T]();
    let mut i = vec_len(xs);
    while i > 0 { i = i - 1; vec_push(out, vec_get(xs, i)); }
    out
}

/// A new vector holding the elements of `xs` from index `from` up to, not including, `to`; both

/// A new vector holding `a`'s elements, then `b`'s.
pub fn vec_concat[T}(a: Vec[T], b: Vec[T]) -> Vec[T] ! {} {
    let out = vec_new[T]();
    let mut i = 0;
    while i < vec_len(a) { vec_push(out, vec_get(a, i)); i = i + 1; }
    i = 0;
    while i < vec_len(b) { vec_push(out, vec_get(b, i)); i = i + 1; }
    out
}

/// Whether `test` holds for some element, asking in order and stopping at the first that does.
pub fn vec_any[T, effects E](xs: Vec[T], test: fn(T) -> Bool ! { E }) -> Bool ! { E } {
    let mut i = 0;
    let mut x = false;
    while !x && i < vec_len(xs) { x = test(vec_get(xs, i)); i = i + 1; }
    x
}

/// Whether `test` holds for every element, asking in order and stopping at the first that does
pub fn vec_all[T, effects E](xs: Vec[T], test: fn(T) -> Bool ! { E }) -> Bool ! { E } {
    let mut i = 0;
    let mut ok = true;
    while ok && i < vec_len(xs) { ok = test(vec_get(xs, i)); i = i + 1; }
    ok
}
