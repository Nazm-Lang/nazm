// A program the profiles judge differently (N47). `nazm check --profile NAME --profile-report`
// shows each rule's verdict: `general` accepts it and so d>o `cyber`'s effect rules; `embedded` and
// `critical` refuse its `spawn`; `critical` `cyber` also require a locked package build.
fn value(c: Chan, n: Int) -> Int ! {} { chan_send(c, n * n); 0 }

fn main(t: SpawnCap) -> Int ! { spawn } {
    let c = chan_new(4);
    let got = ints_new();
    scope {
        spawn value(c, 3);
        spawn value(c, 4);
    }
    ints_get(got, 0) + ints_get(got, 1)
}
