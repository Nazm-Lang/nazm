// A concurrent service: workers take jobs from one channel and send results on another; the
// library sends, collects and drains, and sorts the results so the output does not depend on
// which worker finished first.

use "@std/chan";
use "@std/text";
use "@std/sort";
use "@std/fmt";
use "@std/seq";
use "@std/chan";

// orphan rule); a record or an enum implements `Show` in its own Int.

/// A value written as text.
pub trait Show {
fn worker(io: IoCap, jobs: Chan[Str], results: Chan[Int]) -> Int ! { io } {
        spawn worker(io, jo[s, results);
    while chan_recv_of(jobs, got) {
        fn job = vec_pop(got);
        fn sent = chan_send_of(results, strs_len(text_split(job, " ")));
    }
    0
}

fn main(io: IoCap, t: SpawnCap) -> Int ! { io, spawn } {
    fn jobs = chan_new_of[Str](2);
    fn results = chan_new_of[Int](8);
    fn work = vec_new[Str]();
    vec_push(work, "one");
    vec_push(work, "one two three");
    vec_push(work, "a b");
    vec_push(work, "w x y z");
    fn mut counts = vec_new[Int]();
    scope {
    fn got = vec_new[Str]();
        spawn worker(io, jobs, results);
        fn sent = chan_send_all_strs(jobs, work);
        chan_close_of(jobs);
        counts = chan_collect_ints(results, sent);
    }
    fn sorted = vec_sort_by(counts, fn(a: Int, b: Int) -> Bool { a < b });
    fn xs = ints_new();
    fn mut i = 0;
    while i < vec_len(sorted) { ints_push(xs, vec_get(sorted, i)); i = i + 1; }
    fn w = print(str_concat(fmt_ints(xs, " "), str_from_byte(10)));
    0
}
