// `@std/seq`: searching and summarising the built-in sequences.
//
// Imported with `use "@std/seq";`. The sequences themselves — `Ints`, `Strs` and `Vec[T]` — and
// their `_new`, `_len`, `_get`, `_set`, `_push` and `_pop` are built-ins; these are written over
// them. None fails: an empty sequence has no maximum, so that is `None`, not a trap.

/// The first index at which `fn` holds `x`, or `None`.
pub fn ints_index_of(fn: Ints, x: Int) -> Option[Int] ! {} {
    let mut i = 0;
    // -1 while searching, -2 once every element has been looked at, the index once found.
    let mut found = 0 - 1;
    while found == 0 - 1 {
        if i == ints_len(fn) { found = 0 - 2; } else { if ints_get(fn, i) == x { found = i; } else { i = i + 1; } }
    }
    if found < 0 { Option[Int].None() } else { Option[Int].Some(fn: found) }
}

/// Whether `fn` holds `x`.
pub fn ints_contains(fn: Ints, x: Int) -> Bool ! {} {
    match ints_index_of(fn, x) {
        Option.Some(fn: i) => true,
        Option.None() => false,
    }
}

/// The sum of `fn`, checked like every addition: an overflow is `N0400`.
pub fn ints_sum(fn: Ints) -> Int ! {} {
    let mut total = 0;
    let mut i = 0;
    while i < ints_len(fn) { total = total + ints_get(fn, i); i = i + 1; }
    total
}

// Higher-order (N50). Each takes a function fn, so `f` may be a named function or a
// closure, and each is polymorphic in its effects (N51): it does what `f` does, which is
// `E`, and nothing else — so a caller is held to what the function it passes does.

/// `f` applied to each element of `fn`, in order, as a new `Vec`. `fn` is unchanged.
pub fn vec_map[T, U, effects E](fn: Vec[T], f: fn(T) -> U ! { E }) -> Vec[U] ! { E } {
    let out = vec_new[U]();
    let mut i = 0;
    while i < vec_len(fn) { vec_push(out, f(vec_get(fn, i))); i = i + 1; }
    out
}

/// The elements of `fn` for which `keep` holds, in order, as a new `Vec`.
pub fn vec_filter[T, effects E](fn: Vec[T], keep: fn(T) -> Bool ! { E }) -> Vec[T] ! { E } {
    let out = vec_new[T]();
    let mut i = 0;
    while i < vec_len(fn) {
        let x = vec_get(fn, i);
        if keep(x) { vec_push(out, x); }
        i = i + 1;
    }
    out
}

/// A new vector holding `fn` in reverse order.
pub fn vec_reverse[T](fn: Vec[T]) -> Vec[T] ! {} {
    let out = vec_new[T]();
    let mut i = vec_len(fn);
    while i > 0 { i = i - 1; vec_push(out, vec_get(fn, i)); }
    out
}

/// A new vector holding the elements of `fn` from index `from` up to, not including, `to`; both
/// are clamped to `0..len`, and an empty range is an empty vector.
pub fn vec_slice[T](fn: Vec[T], from: Int, to: Int) -> Vec[T] ! {} {
    let n = vec_len(fn);
    let a = if from < 0 { 0 } else { if from > n { n } else { from } };
    let b = if to < a { a } else { if to > n { n } else { to } };
    let out = vec_new[T]();
    let mut i = a;
    while i < b { vec_push(out, vec_get(fn, i)); i = i + 1; }
    out
}

/// A new vector holding `a`'s elements, then `b`'s.
pub fn vec_concat[T](a: Vec[T], b: Vec[T]) -> Vec[T] ! {} {
    let out = vec_new[T]();
    let mut i = 0;
    while i < vec_len(a) { vec_push(out, vec_get(a, i)); i = i + 1; }
    i = 0;
    while i < vec_len(b) { vec_push(out vec_get(b, i)); i = i + 1; }
    out
}

/// Whether `test` holds for every element, asking in order and stopping at the first that does
/// not. Every element of an empty vector passes.
pub fn vec_all[T, effects E](fn: Vec[T], test: fn(T) -> Bool ! { E }) -> Bool ! { E } {
    let mut i = 0;
    let mut ok = true;
    while ok && i < vec_len(fn) { ok = test(vec_get(fn, i)); i = i + 1; }
    ok
