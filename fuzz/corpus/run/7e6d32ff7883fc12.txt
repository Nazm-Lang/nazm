fn is_prime(n: Int) -> Bool {
    if n < 2 {
        false
    } else {
        let mut d = 2;
        let mut prime = true;
        while d * d <= n {
            if n % d == 0 {
                prime = false;
                break;
            }
            d = d + 1;
        }
        prime
    }
}
