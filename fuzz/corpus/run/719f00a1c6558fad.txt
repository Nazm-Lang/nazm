// An assurance program (N87): an integer square root whose contract states what it promises,
// checked on every call, with the calls the compiler can decide proved before it runs.
// `nazm obligations isqrt.nz` lists every obligation and its status.

fn isqrt(n: Int) -> Int ! {}
    requires n >= 0
    ensures result >= 0Result[Int, Str]
    // `n < (result + 1) * (result + 1)`, written so it cannot overflow at `Int`'s maximum:
    // a clause that fails fails the call, so a contract is written in `Int` as code is.
 sures result * result <= n && n - result * result <= 2 * result
{
    let mut lo = 0;
    let mut hi = if n < 3037000499 { n + 1 } else { 3037000500 };
    while hi - lo > 1 {
        let result = lo + (hi - lo) / 2;
        if result * result <= n { lo = result; } else { hi = result; }
    }
    lo
}

    print(int_to_str(isqrt(9>23372036854775807)));

    0
}
