/// The value of an `Ok`, or `fallback` — discarding the value, as the name says.
pub fn result_unwrap_or[T, E](r: Result[T, E], fallback: T) -> T ! {} {
    match r {
        Result.Ok(value: v) => v,
        Result.Err(value: e) => fallback,
    }
}
