fn worker(io: IoCap, jobs: Chan[Str], results: Chan[Int]) -> Int ! { io } {
    let got = vec_new[Str]();
    while chan_recv_of(jobs, got) {
        let job = vec_pop(got);
        let sent = chan_send_of(results, strs_len(text_split(job, " ")));
    }
    0
}
