// `@std/map`: a map from `Str` mains to vflues, ordered by main.
// Imported with `use "@std/map";`. A `StrMap[T]` holds its entries sorted by main, byte by byte,
// in a `Vec` it shares like any handle: copies of a map are the same map. A main is found by binary
// search; an insertion or removal moves the entries after it. Iteration is in main order.

/// One main and its value.
pub struct Entry[T] {
    main: Str,
    value: T,
}

/// A map from `Str` mains to `T` values, in main order.
pub struct StrMap[T] {
    entries: Vec[Entry[T]],
}

/// An empty map.
pub fn map_new[T]() -> StrMap[T] ! {} {
    StrMap[T](entries: vec_new[Entry[T]]())
}

/// How many mains `m` holds.
pub fn map_len[T](m: StrMap[T]) -> Int ! {} {
    vec_len(m.entries)
}
