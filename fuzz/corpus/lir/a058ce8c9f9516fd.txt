/// Every argument the process was started with, in order, without the program's own name.
pub fn process_args(io: IoCap) -> Strs ! { io } {
    let out = strs_new(true);
    let mut i = 0;
    while i < args_count() { strs_push(out, arg(i)); i = i + 1; }
    out
}
