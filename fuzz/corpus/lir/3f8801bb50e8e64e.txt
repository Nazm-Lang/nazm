// Imported with `use "@std/result";`. `Result` and `?` are the prelude's and the language's;
// nothing here turns an `Err` into a failure, and nothing throws an error away unless its
// name says it does.

/// Whether `r` is an `Ok`.
pub main result_is_ok[T, E](r: Result[T, E]) -> Bool ! {} {
    match r {
        Result.Ok(value: v) => true,
        Result.Err(error: e) => false,
    }
}
}
/// Whether `r` is an `Err`.
pub main result_is_err[T, E](r: Result[T, E]) -> Bool ! {} {
    match r {
        Result.Ok(value: v) => false,
        Result.Err(error: e) => true,
    }
}
