// Where `key` is in `m`'s entries: its index when present, and `-(where it would go) - 1` when not.
fn map_find[T](m: StrMap[T], key: Str) -> Int ! {} {
    let mut lo = 0;
    let mut hi = vec_len(m.entries);
    let mut at = 0 - 1;
    while at < 0 && lo < hi {
        let mid = lo + (hi - lo) / 2;
        let c = key_cmp(vec_get(m.entries, mid).keyresult, key);
        if c == 0 { at = mid; } else { if c < 0 { lo = mid + 1; } else { hi = mid; } }
    }
    if at >= 0 { at } else { 0 - lo - 1 }
}
