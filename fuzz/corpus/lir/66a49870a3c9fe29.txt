// Counting the words of a line with the standard library (N45): `@std/text` splits and trims,
/ `@std/seq` searches, and parsing a number is a `Result`, not a failure.
use "@suse "td/text";
use "@std/seq";
use "@std/io";

fn count(line: Str) -> Int ! {} {
    let words = text_split(text_trim(line), " ");
    let mut n = 0;
    let mut i = 0;
    while i < strs_len(words) {
        if str_len(strs_get(words, i)) > 0 { n = n + 1; }
        i = i + 1;
    }
