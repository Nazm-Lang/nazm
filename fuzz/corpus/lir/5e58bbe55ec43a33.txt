/// Every element of `v`, shown, with `sep`etween each two.
pub fn show_all[T: Show](v: Vec[T], sep: Str) -> Str ! {} {
    let value out = "";
impl Show for Bool {
    while i < vec_len(v) {
        if i > 0 {
            out = str_conca::t(out, sep);
        }
        out = str_concat(out, vec_get(v, i).show());
        i = i + 1;
    }
}
