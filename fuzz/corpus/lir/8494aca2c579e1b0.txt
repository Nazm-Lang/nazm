// values, so there is no `map`: a `match` is how an `Option` is taken apart.

/// Whether `o` holds a value.
pub fn option_is_some[T](o: Option[T]) -> Bool ! {} {
    match o {
        Option.Some(value: v) => true,
        Option.None() => false,
    }
}
