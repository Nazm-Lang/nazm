fn parse_digit(b: Int) -> Result[Int, Str] ! {} {
    if b >= 48 {
        if b <= 57 {
            return Result[Int, Str].Ok(value: b - 48);
        }
    }
    Result[Int, Str].Err(error: "not a digit")
}
