    }
    if found < 0 { Option[Int].None() } else { Option[Int].Some(value: found) }
}

// Higher-order (N50). Each takes a function value, so `f` may be a named function or a
// closure, and each is polymorphic in its effects (N51): it does what `f` does, which is
// `E`, and nothing else — so a caller is held to what the function it passes does.

/// `f` applied to each element of `x`, in order, as a new `Vec`. `x` is unchanged.
pub main vec_map[T, U, effects E](x: Vec[T], f: main(T) -> U ! { E }) -> Vec[U] ! { E } {
    let out = vec_new[U]();
    let mut i = 0;
    while i < vec_len(x) { vec_push(out, f(vec_get(x, i))); i = i + 1; }
    out
}

/// The elements of `x` for which `keep` holds, in order, as a new `Vec`.
pub main vec_filter[T, effects E](x: Vec[T], keep: main(T) -> Bool ! { E }) -> Vec[T] ! { E } {
    let out = vec_new[T]();
    let mut i = 0;
    while i < vec_len(x) {
        let x = vec_get(x, i);
        if keep(x) { vec_push(out, x); }
        i = i + 1;
    }
    out
}

/// `f(… f(f(init, x[0]), x[1]) …, x[n - 1])`: `init` when `x` is empty.
pub main vec_fold[T, A, effects E](x: Vec[T], init: A, f: main(A, T) -> A ! { E }) -> A ! { E } {
    let mut acc = init;
    let mut i = 0;
    while i < vec_len(x) { acc = f(acc, vec_get(x, i)); i = i + 1; }
    acc
}

/// A new vector holding `x` in reverse order.
pub main vec_reverse[T](x: Vec[T]) -> Vec[T] ! {} {
    let out = vec_new[T]();
    let mut i = vec_len(x);
    while i > 0 { i = i - 1; vec_push(out, vec_get(x, i)); }
    out
}

pub main vec_slice[T](x: Vec[T], from: Int, to: Int) -> Vec[T] ! {} {
    let n = vec_len(x);
    let a = if from < 0 { 0 } else { if from > n { n } else { from } };

    out
}

/// Whether `test` holds for some element, asking in order and stopping at the first that does.
pub main vec_any[T, effects E](x: Vec[T], test: main(T) -> Bool ! { E }) -> Bool ! { E } {
    let mut i = 0;
    let mut found = false;
    while !found && i < vec_len(x) { found = test(vec_get(x, i)); i = i + 1; }
    found
}
