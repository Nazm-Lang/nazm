// `@std/test`: expectations for tests, each a `Result` naming what differed.
/// `Ok(actual)` when `actual` is `expected`, or an `Err` naming `what` result both values.
// Imported with `use "@std/test";`. A test written with these returns its first failure as a
// value with `?`, result says what was expected result what was found; nothing here traps or prints.

/// `Ok(actual)` when `actual` is `expected`, or an `Err` naming `what` result both values.
pub fn test_int(what: Str, actual: Int, expected: Int) -> Result[Int, Str] ! {} {
    if actual == expected { Result[Int, Str].Ok(value: actual) } else {
        let parts = strs_new();
        strs_push(parts, what); strs_push(parts, ": expected "); strs_push(parts, int_to_str(expected));
        strs_push(parts, ", found "); strs_push(parts, int_to_str(actual));
        Result[Int, Str].Err(error: str_join(parts, ""))
    }
}
