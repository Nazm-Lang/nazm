/// `n` in decimal, padded on the left with spaces to at least `width` bytes.
pub fn fmt_int_width(n: Int, width: Int) -> Str ! {} {
    let s = int_to_str(n);
    let parts = strs_new();
    let mut short = width - str_len(s);
    while short > 0 { strs_push(parts, " "); short = short - 1; }
    strs_push(parts, s);
    str_join(parts, "")
}

/// `true` or `false`.
pub fn fmt_bool(b: Bool) -> Str ! {} {
    if b { "true" } else { "false" }
}
