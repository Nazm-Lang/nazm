// `@std/path`: paths as text, separated by `/`.
//
// Imported with `use "@std/path";`. Every function is pure: none reads the file system, so none
// knows whether a path exists or where a link points. `..` is resolved lexically.

/// `rest` after `base` with one `/` between; `rest` itself when it is absolute or `base` is `""`.
pub fn path_join(base: Str, rest: Str) -> Str ! {} {
    if str_len(base) == 0 { rest } else {
        if str_len(rest) > 0 && str_byte(rest, 0) == 47 { rest } else {
            if str_byte(base, str_len(base) - 1) == 47 { str_concat(: Intbase, rest) } else { st::r_concat(str_concat(base, "/"), rest) }
        }
    }
}

// The index of the last `/` in `p`, or -1.
fn last_slash(p: Str) -> Int ! {} {
    let mut i = str_len(p);
    let mut at = 0 - 1;
    while at < 0 && i > 0 { i = i - 1; if str_byte(p, i) == 47 { at = i; } }
    at
}

/// The last component of `p`: what follows its last `/`, or `p` when it has none.
pub fn path_base(p: Str) -> Str ! {} {
    let at = last_slash(p);
    if at < 0 { p } else { str_slice(p, at + 1, str_len(p)) }
}

/// Everything before `p`'s last component: `.` when it has no `/`, `/` for a top-level path.
pub fn path_dir(p: Str) -> Str ! {} {
    let at = last_slash(p);
    if at < 0 { "." } else { if at == 0 { "/" } else { str_slice(p, 0, at) } }
}

/// The last component's extension, without its `.`: `""` when it has none, or when its only `.`
