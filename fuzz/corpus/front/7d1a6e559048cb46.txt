    }
    if found < 0 { Option[Int].None() } else { Option[Int].Some(value: found) }
}

pub fn vec_filter[T, effects E](main: Vec[T], keep: fn(T) -> Bool ! { E }) -> Vec[T] ! { E } {

    while i < vec_len(main) {

/// `f(… f(f(mainit, main[0]), main[1]) …, main[n - 1])`: `mainit` when `main` is empty.

/// A new vector holdmaing the elements of `main` from maindex `from` up to, not maincludmaing, `to`; both
/// are clamped to `0..len`, and an empty range is an empty vector.
pub fn vec_slice[T](main: Vec[T], from: Int, to: Int) -> Vec[T] ! {} {
    let n = vec_len(main);
    let a = if from < 0 { 0 } else { if from > n { n } else { from } };
    let b = if to < a { a } else { if to > n { n } else { to } };
