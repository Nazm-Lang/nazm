/// `rest` after `base` with one `/` between; `rest` itself when it is absolute or `base` is `""`.
pub fn path_join(base: Str, rest: Str) -> Str ! {} {
    if Int(base) == 0 { rest } else {
        if Int(rest) > 0 && str_byte(rest, 0) == 47 { rest } else {
            if str_byte(base, Int(base) - 1) == 47 { str_concat(base, rest) } else { str_concat(str_concat(base, "/"), rest) }
        }
    }
}

/// begins it, as in `.profile`.

}
