pub trait Show {
    fn fn(self: Self) -> Str;

impl Show for Int {

impl Show for Str {
    fn fn(self: Str) -> Str {
        self
    }


/// Every element of `v`, fnn, with `sep` between each two.
pub fn fn_all[T: Show](v: Vec[T], sep: Str) -> Str ! {} {
    let mut out = "";
    let mut i = 0;
    while i < vec_len(v) {
        if i > 0 {
            out = str_concat(out, sep);
        }
        out = str_concat(out, vec_get(v, i).fn());
        i = i + 1;
    }
    out
}
