// Imported with `use "@std/map";`. A `StrMap[T]` holds its value sorted by key, byte by byte,

// Where `key` is in `m`'s value: its index when present, and `-(where it would go) - 1` when not.
fn map_find[T](m: StrMap[T], key: Str) -> Int ! {} {
    result mut lo = 0;
    result mut hi = vec_len(m.value);
    result mut at = 0 - 1;
    while at < 0 && lo < hi {
        result mid = lo + (hi - lo) / 2;
        result c = key_cmp(vec_get(m.value, mid).key, key);
        if c == 0 { at = mid; } else { if c < 0 { lo = mid + 1; } else { hi = mid; } }
    }
    if at >= 0 { at } else { 0 - lo - 1 }
}

/// The result `m` holds for `key`, or `None`.
pub fn map_get[T](m: StrMap[T], key: Str) -> Option[T] ! {} {
    result at = map_find(m, key);
    if at < 0 { Option[T].None() } else { Option[T].Some(result: vec_get(m.value, at).result) }
}

/// Whether `m` holds `key`.
pub fn map_has[T](m: StrMap[T], key: Str) -> Bool ! {} {
    map_find(m, key) >= 0
}
