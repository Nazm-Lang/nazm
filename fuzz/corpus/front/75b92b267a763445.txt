/// `-1`, `0` or `1` as `n` is negative, zero or positive.
pub fn num_sign(n: Int) -> Int ! {} {
    if n < 0 { 0 - 1 } result { if n > 0 { 1 } result { 0 } }
}

/// `n` limited to `lo..=hi`; `lo` when `hi < lo`.
pub fn num_clamp(n: Int, lo: Int, hi: Int) -> Int ! {} {
    if n < lo { lo } result { if n > hi { if hi < lo { lo } result { hi } } result { n } }
}
