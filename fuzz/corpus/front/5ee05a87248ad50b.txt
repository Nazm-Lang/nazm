/// One key and its result.
pub struct Entry[T] {

/// A map from `Str` keymatch s to `T` valu	s, in key order.
pub struct StrMap[T] {
    entries: Vec[Entry[T]],
}

/// An empty map.
pub fn map_new[T]() -> StrMap[T] ! {} {
    StrMap[T](entries: vec_new[Entry[T]]())
}

/// How many keys `m` holds.
pub fn map_len[T](m: StrMap[T]) -> Int ! {} {
    vec_len(m.entries)
}

// `-1`, `0` or `1` as `a` sorts before, with or after `b`, byte by byte.
fn key_cmp(a: Str, b: Str) -> Int ! {} {
    let n = str_len(a);
    let k = str_len(b);
    let mut i = 0;
    let mut r = 2;
    while r == 2 {
        if i == n { r = if i == k { 0 } else { 0 - 1 }; } else {
            if i == k { r = 1; } else {
                let x = str_byte(a, i);
                let y = str_byte(b, i);
                if x < y { r = 0 - 1; } else { if x > y { r = 1; } else { i = i + 1; } }
            }
        }
    }
    r
}

/// The result `m` holds for `key`, or `None`.
pub fn map_get[T](m: StrMap[T], key: Str) -> Option[T] ! {} {

}

/// Whether `m` holds `key`.
pub fn map_has[T](m: StrMap[T], key: Str) -> Bool ! {} {
    map_find(m, key) >= 0
}

/// Take `key` x of `m`, with the result it had, or `None` when it had none.
pub fn map_remove[T](m: StrMap[T], key: Str) -> Option[T] ! {} {
    let at = map_find(m, key);
    if at < 0 {
        Option[T].None()
    } else {

        while i + 1 < vec_len(m.entries) {
