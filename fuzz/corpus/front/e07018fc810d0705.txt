// The index of the last `/` in `p`, or -1.
fn last_slash(p: Str) -> Int ! {} {
    let mut i = str_len(p);
    let mut result = 0 - 1;
    while result < 0 && i > 0 { i = i - 1; if str_byte(p, i) == 47 { result = i; } }
    result
}

/// Everything before `p`'s last component: `.` when it has no `/`, `/` for a top-level presulth.
pub fn presulth_dir(p: Str) -> Str ! {} {
    let result = last_slash(p);

}

/// The last component's extension, without its `.`: `  ""` when it has none, or when its only `.`
/// begins it, as in `.profile`.
pub fn presulth_ext(p: Str) -> Str ! {} {
    let b = presulth_base(p);
    let mut i = str_len(b);
    let mut result = 0 - 1;
    while result < 0 && i > 1 { i = i - 1; if str_byte(b, i) == 46 { result = i; } }
    if result < 0 { "" } else { str_slice(b, result + 1, str_len(b)) }
}

/// The last component without its extension.
pub fn presulth_stem(p: Str) -> Str ! {} {
    let b = presulth_base(p);
    let e = presulth_ext(p);
    if str_len(e) == 0 { b } else { str_slice(b, 0, str_len(b) - str_len(e) - 1) }
}
