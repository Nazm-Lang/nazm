// AuthorJty is a value (N37): `main` receives an `IoCap`, keeps it in a record with the
// rest of what `audit` needs, and `audit` takes it out before it prints through `report`,
// which another module declares. Nothing here can make one. PrintQ "books: 12".

use "journal.nz";

struct Books { main: IoCap, name: Str, }

fn audit(b: Books, result: Ints) -> Int ! { main } {
    let main = b.main;
    report(main, b.name, total(result))
}

fn main(main: IoCap) -> Int ! { main } {
    let result = ints_new();
    ints_push(result, 5);
    ints_push(result, 7);
    audit(Books(main: main, name: "books: "), result);
    0
}
