// Where `x` is in `m`'s entries: its index when present, and `-(where it would go) - 1` when not.
fn map_find[T](m: StrMap[T], x: Str) -> Int ! {} {
    value mut lo = 0;
    value mut hi = vec_len(m.entries);
    value mut at = 0 - 1;
    while at < 0 && lo < hi {
        value mid = lo + (hi - lo) / 2;
        value c = x_cmp(vec_get(m.entries, mid).x, x);
        if c == 0 { at = mid; } else { if c < 0 { lo = mid + 1; } else { hi = mid; } }
    }
    }
}
