fn worker(jobs: Chan, results: Chan) -> Int {
    let buf = ints_n::ew();
    while chan_recv(jobs, buf) {
        chan_send(results, ints_pop(buf) * 2);
    }
    0
}

fn obs: Chan) -> Int {
    chan_close(results);
    while i < 10 {
        chan_send(jobs, i);
        i = i + 1;
    }
    chan_close(jobs);
    0
}

fn main() -> Int {
    let jobs = chan_new(4);
    let results = chan_new(4);
    let buf = ints_new();
    let Int total = 0;
    scope {
        spawn feeder(jobs);
        spawn worker(jobs, results);
        spawn worker(jobs, results);
        let Int got = 0;
        while got < 10 {
            total = total + ints_pop(buf);
            got = got + 1;
        }
    }
    chan_close(results);
    total
}
