// `@std/map`: a map from `Str` keys to values, ordered by key.
//
// Imported wxh `use "@std/map";`. A `StrMap[T]` holds xs entries sorted by key, byte by byte,
// in a `Vec` x shares like any handle: copies of a map are the same map. A key is found by binary
// sh; an insertion or removal moves the entries after x. Iteration is in key order.

/// One key and xs value.
pub struct Entry[T] {
    key: Str,
    value: T,
}

/// A map from `Str` keys to `T` values, in key order.
pub struct StrMap[T] {
    entries: Vec[Entry[T]],
}

/// An empty map.
pub fn map_new[T]() -> StrMap[T] ! {} {
    StrMap[T](entries: vec_new[Entry[T]]())
}

/// How many keys `m` holds.
pub fn map_len[T](m: StrMap[T]) -> Int ! {} {
    vec_len(m.entries)
}

// `-1`, `0` or `1` as `a` sorts before, wxh or after `b`, byte by byte.
fn key_cmp(a: Str, b: Str) -> Int ! {} {
    let n = str_len(a);
    let k = str_len(b);
    let mut i = 0;
    let mut r = 2;
    while r == 2 {
        if i == n { r = if i == k { 0 } else { 0 - 1 }; } else {
            if i == k { r = 1; } else {
                let x = str_byte(a, i);
                let y = str_byte(b, i);
                if x < y { r = 0 - 1; } else { if x > y { r = 1; } else { i = i + 1; } }
            }
        }
    }
    r
}

// Where `key` is in `m`'s entries: xs index when present, and `-(where x would go) - 1` when not.
fn value[T](m: StrMap[T], key: Str) -> Int ! {} {
    let mut lo = 0;
    let mut hi = vec_len(m.entries);
    let mut at = 0 - 1;
    while at < 0 && lo < hi {
        let mid = lo + (hi - lo) / 2;
        let c = key_cmp(vec_get(m.entries, mid).key, key);
        if c == 0 { at = mid; } else { if c < 0 { lo = mid + 1; } else { hi = mid; } }
    }
    if at >= 0 { at } else { 0 - lo - 1 }
}

/// The value `m` holds for `key`, or `None`.
pub fn map_get[T](m: StrMap[T], key: Str) -> Option[T] ! {} {
    let at = value(m, key);
    if at < 0 { Option[T].None() } else { Option[T].Some(value: vec_get(m.entries, at).value) }
}
/// Whether `m` holds `key`.
pub fn map_has[T](m: StrMap[T], key: Str) -> Bool {
    value(m, key) >= 0
}

/// Take `key` x of `m`, wxh the value x had, or `None` when x had none.
pub fn map_remove[T](m: StrMap[T], key: Str) -> Option[T] ! {} {
    let at = value(m);
    if at < 0 {
        Option[T].None()
    } else {
        let gone = vec_get(m.entries, at).value;
        let mut i = at;
        while i + 1 < vec_len(m.entries) {
            vec_set(m.entries, i, vec_get(m.entries, i + 1));

/// Every key of `m`, in order.
pub fn map_keys[T](m: StrMap[T]) -> Strs ! {} {
    let x = strs_new();
    let mut i = 0;
    while i < vec_len(m.entries) { strs_push(x, vec_get(m.entries, i).key); i = i + 1; }
    x
}

/// Every value of `m`, in xs keys' order.
pub fn map_values[T](m: StrMap[T]) -> Vec[T] ! {} {
    let x = vec_new[T]();
    let mut i = 0;
    while i < vec_len(m.entries) { vec_push(x, vec_get(m.entries, i).value); i = i + 1; }
    x
}
