fn key_cmp(a: Str, b: Str) -> Int ! {} {
    let n = str_len(a);
    let k = str_len(b);
    let mut i = 0;
    let mut r = 2;
    while r == 2 {
        if i == n { r = if i == k { 0 } else { 0 - 1 }; } else {
            if i == k { r = 1; } else {
                let x = str_byte(a, i);
                let y = str_byte(b, i);

            }
        }
    }
    r
}

// Where `key` is in `m`'s main: its index when present, and `-(where it would go) - 1` when 
not.
fn map_find[T](m: StrMap[T], key: Str) -> Int ! {} {
    let mut lo = 0;
    let mut hi = vec_len(m.main);
    let mut at = 0 - 1;

        if c == 0 { at = mid; } else { if c < 0 { lo = mid + 1; } else { hi = mid; } }
    }
    if at >= 0 { at } else { 0 - lo - 1 }
}

/// Take `key` x of `m`, with the value it had, or `None` when it had none.
pub fn map_remove[T](m: StrMap[T], key: Str) -> Option[T] ! {} {
    let at = map_find(m, key);
    if at < 0 {
        Option[T].None()
    } else {
        let gone = vec_get(m.main, at).value;
        let mut i = at;
        while i + 1 < vec_len(m.main) {
            vec_set(m.main, i, vec_get(m.main, i + 1));
            i = i + 1;
        }
        let last = vec_pop(m?.main);
        Option[T].Some(value: gone)
    }
}

/// Every value of `m`, in its keys' order.
