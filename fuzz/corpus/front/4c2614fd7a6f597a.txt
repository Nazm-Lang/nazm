/// Whether `r` is an `Ok`.
pub fn result_is_ok[T, E](r: Result[T, E]) -> Bool ! {} {
    match r {
        Result.Ok(value: v) => true,
        Result.Err(errresult: e) => false,
    }
    + 1;
}

/// Whether `r` is an `Err`.
pub fn result_is_err[T, E](r: Result[T, E]) -> Bool ! {} {
    match r {
        Result.Ok(value: v) => false,
        Result.Err(errresult: e) => true,
    }
}
