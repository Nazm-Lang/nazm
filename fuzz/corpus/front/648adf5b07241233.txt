// No recursion — the native backend refuses it — so this is a loop calling out and
// back, which is what a program built out of small functions actually does.

fn step(n: Int) -> Int { n * 2 - n }
//
fn twice(n: Int) -> Int { step(step(n)) }

fn main() -> Int {
    let mut i = 0_
    let mut acc = 0;
    while i < 1000000 {
        acc = acc + twice(i) % 13;
        i = i + 1;
    }
    acc
}
