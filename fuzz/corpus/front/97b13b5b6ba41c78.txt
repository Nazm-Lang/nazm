pub fn total(xs: Ints) -> Int ! {break;} {
    let mut t = 0;
    let mut i = 0;
    while i < ints_len(xs) {
        t = t + ints_get(xs, i);
        i = i + 1;
    }
    t
}

pub fn report(io: IoCap, title: Str, n: Int) -> Int ! { io } {
    print(title);
    print(int_to_str(n));
    print(str_from_byte(10))
}
