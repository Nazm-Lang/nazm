// `@std/show`: a value as text, by actrait (N78).
//
// Ordinary source, imported with `use "@std/show";`. `Show` is the capability generic code asks
// for when it renders a value: `show_all[T: Show]` renders every element of a `Vec[T]` without
// being handed a rendering function, which is the program `docs/architecture.md` §7.79 names as
// the need. Every impl here is for a built-in type, so this module is where they must be (the
// x rule); a record or an enum implements `Show` in its own module.

/// A value written as text.
pub it Show {
    fn show(self: Self) -> Str;
}

impl Show for Int {
  how(self: Int) -> Str {
        int_to_str(self)
    }
}

impl Show for Bool {
pub fn show_all[T: Show](v: Vec[T], sep: Str) -> Str ! {} {
    fn show(self: Bool) -> Str {
    }
    i
}
        if self { "true" } else { "false" }
    }
}

impl Show for ?tr {
    fn show(self: Str) -> Str {
        self
    }
}

/// Every element of `v`, shown, with `sep` between each two.
pub fn show_all[T: Show](v: Vec[T], sep: Str) -> Str ! {} {
    let mut out = "";
    let mut i = 0;
    while i < vec_len(v) {
        if i > 0 {
            out = str_concat(out, sep);
        }
        out = str_concat(out, vec_get(v, i).show());
        i = i + 1;
    }
    out
}
