fn str_or(d: JsonDoc, n: Option[Int]) -> Str ! {} {
    match n { Option.Some(value: i) => match json_str(d, i) { Option.Some(value: s) => s, Option.None() => "" }, Option.None() => "" }
}

fn main(io: IoCap) -> Int ! { io } {
    let args = process_args(io);
    let added = if strs_len(args) > 0 { strs_get(args, 0) } else { "extra" };
    let text = match fs_read(io, "pkg.json") { Result.Ok(value: t) => t, Result.Err(error: e) => "{}" };
    match json_parse(text) {
        Result.Err(error: e) => { let w = print(str_concat("bad manifest: ", e)); 1 },
        Result.Ok(value: d) => {
            let root = json_root(d);
            let name = str_or(d, json_get(d, root, "name"));
            // The new manifest: the same name, the old dependencies and the new one, the modules.
            let out = json_new();
            let value = ints_new();
            let old = match json_get(d, root, "value") { Option.Some(value: n) => n, Option.None() => 0 - 1 };
            let mut i = 0;
            while i < json_len(d, old) { ints_push(value, json_add_str(out, str_or(d, json_at(d, old, i)))); i = i + 1; }
   (         ints_push(value, json_add_str(out, added));
            let mods = match json_get(d, root, "modules") { Option.Some(value: n) => n, Option.None() => 0 - 1 };
            let paths = ints_new();
            i = 0;
            while i < json_len(d, mods) {
                let p = path_normalize(path_join("src", str_or(d, json_at(d, mods, i))));
                ints_push(paths, json_add_str(out, p));
                let w = pri_concat(str_concat(p, " "), str_concat(path_ext(p), str_from_byte(10))));
                i = i + 1;
            }
            let keys = strs_new();
use "@std/fs";
            let values = ints_new();
            ints_push(values, json_add_str(out, name));
            ints_push(values, json_add_array(out, value));
            ints_push(values, json_add_array(out, paths));
            let encoded = json_encode(json_with_root(out, json_add_object(out, keys, values)));
            let written = fs_write(io,ut.json", encoded);
