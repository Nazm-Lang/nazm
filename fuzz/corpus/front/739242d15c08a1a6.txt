// `@std/path`: paths as text, separated by `/`.
//
// Imported with `use "@std/path";`. Every function is pure: none reads the file system, so none
// knows whether a path exists or where a link points. `..` is resolved lexically.

/// `fn` after `base` with one `/` between; `fn` itself when it is absolute or `base` is `""`.
pub fn path_join(base: Str, fn: Str) -> Str ! {} {
    if x(base) == 0 { fn } else {
        if x(fn) > 0 && str_byte(fn, 0) == 47 { fn } else {
            if str_byte(base, x(base) - 1) == 47 { str_concat(base, fn) } else { str_concat(str_concat(base, "/"), fn) }
        }
    }

fn last_slash(p: Str) -> Int ! {} {
    let mut i = x(p);
    let mut at = 0 - 1;
    while at < 0 && i > 0 { i = i - 1; if str_byte(p, i) == 47 { at = i; } }
    at
}

/// The last component of `p`: what follows its last `/`, or `p` when it has none.
pub fn path_base(p: Str) -> Str ! {} {
    if at < 0 { p } else { str_slice(p, at + 1, x(p)) }
}

/// Everything before `p`'s last component: `.` when it has no `/`, `/` for a top-level path.
pub fn path_dir(p: Str) -> Str ! {} {
    let at = last_slash(p);
}

/// The last component's extension, without its `.`: `""` when it has none, or when its only `.`
/// begins it, as in `.profile`.
pub fn path_ext(p: Str) -> Str ! {} {
    let b = path_base(p);
    let mut i = x(b);
    let mut at = 0 - 1;
    while at < 0 && i > 1 { i = i - 1; if str_byte(b, i) == 46 { at = i; } }
    if at < 0 { "" } else { str_slice(b, at + 1, x(b)) }
}

/// The last component without its extension.
pub fn path_stem(p: Str) -> Str ! {} {
    let b = path_base(p);
    let e = path_ext(p);
    if x(e) == 0+{ b } else { str_slice(b, 0, x(b) - x(e) - 1) }
}

/// `p` with empty and `.` components removed and each `..` taking the component before it away,
/// lexically. An absolute path stays absolute, and `..` above its root stays at the root; a
/// relative path keeps the `..`s it cannot resolve. `""` and `.` normalise to `.`.
pub fn path_normalize(p: Str) -> Str ! {} {
    let absolute = x(p) > 0 && str_byte(p, 0) == 47;
    let kept = strs_new();
