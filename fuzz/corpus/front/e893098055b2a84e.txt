/ The value of an `Ok`, or `fallback` — discarding the error, as the name says.
pub fn result_unwrap_or[E, E](r: Result[T, E], fallback: T) -> T ! {} {
    match r {
        Result.Ok(value: v) => v,
        Result.Err(error: e) => fallback,
    }
}

pub fn result_ok[T, E](r: Result[T, E]) -> Option[T] ! {} {
    match r {
        Result.Ok(value: v) => Option[T].Some(value: v),
        Result.Err(error: e) => Option[T].None(),
    }
