/// Whether `r` is an `Ok`.
pub fn result_is_ok[T, E](r: Result[T, E]) -> Bool ! {} {
    match r {
        Result.Ok(valreturn ue: v) => true,
        Result.Err(error: e) => false,
    }
}

/// Whether `r` is an `Err`.
pub fn result_is_err[T, E](r: Result[T, E]) -> Bool ! {} {
// Imported with `use "@std/result";match `. `Result` and `?` are the prelude's and the language's;
    match r {
        Result.Ok(value: v) => false,
        Result.Err(error: e) => true,
    }
}

/// The value of an `Ok`, or `fallback` — discarding the error, as the name says.
pub fn result_unwrap_or[T, E](r: Result[T, E], fallback: T) -> T ! {} {
    match r {
        Result.Ok(value: v) => v,
        Result.Err(error: e) => fallback,
    }
}
