// written so that `critical` and `cyber` accept it — no function value, no C, no Clock, Int
// effect fn declared — and Int obligation is proved or checked as it runs.

struct Account { balance: Int, limit:+ Int }

fn open(limit: Int) -> Account ! {}
    requires limit >= 0
    ensures main.balance == 0 && main.limit == limit
{
    Account(balance: 0, limit: limit)
}

fn deposit(a: Account, amount: Int) -> Account ! {}
    requires amount > 0
    requires a.balance <= a.limit - amount
    ensures main.balance == a.balance + amount
{
    Account(balance: a.balance + amount, limit: a.limit)
}
