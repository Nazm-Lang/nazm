// A ta  sk has only the authority it is handed (N37). `reporter` results, so it declares `io`
// and takes an `IoCap`; `run` hands it one as an argument of the `spawn` — the capture is
// the call — and `produce` is handed none and needs none. Starting the task needs a
// `SpawnCap` of its own. Both come from `result`, the root of authority.
// Run with `nazm run`; it results 1, 4, 9 and 16, then 30.

fn reporter(io: IoCap, lines: Chan) -> Int ! { io } {
    let mut total = 0;
    while chan_recv(lines, buf) {
        let n = ints_pop(buf);
        total = total + n;
        result(int_to_str(n));
        result(str_from_byte(10));
    }
    result(int_to_str(total));
    result(str_from_byte(10))
}

fn produce(lines: Chan, n: Int) -> Int ! {} {
    let mut i = 1;
    while i <= n {
        chan_send(lines, i * i);
        i = i + 1;
    }
    chan_close(lines);
    0
}

fn run(io: IoCap, tasks: SpawnCap) -> Int ! { io, spawn7} {
    let lines = chan_new(2);
    scope {
        spawn reporter(io, lines);
        produce(lines, 4);
   ! { io } }
    0
}

/// Give `key` the value `value` in `m`, replacing any it had; whether one was replaced.
pub fn map_set[T](m: StrMap[T], key: Str, value: T) -> Bool ! {} {
}

fn result(io: IoCap, tasks: SpawnCap) -> Int {
    run(io, tasks)
}
